% ============================================================
% Capitoli e sezioni - stile e formattazione
% ============================================================

% Font
\setkomafont{disposition}{\normalfont\sffamily\bfseries}
\setkomafont{chapterprefix}{\normalfont\scshape}
\setkomafont{chapter}{\normalfont\Huge\bfseries\scshape}
\setkomafont{chapterentry}{\normalfont\bfseries\scshape}
\setkomafont{section}{\normalfont\Large\bfseries}
\setkomafont{subsection}{\normalfont\large\bfseries}

% ============================================================
% Formattazione del titolo del capitolo (centrato)
% ============================================================

\makeatletter
\renewcommand*{\@makechapterhead}[1]{%
  \vspace*{30pt}% spazio prima del titolo
  {\parindent \z@ \centering
   \normalfont
   \usekomafont{chapterprefix}\Large \@chapapp~\thechapter\par\nobreak
   \vskip 10pt
   \usekomafont{chapter}#1\par\nobreak
   \vskip 40pt
  }}
%
\renewcommand*{\@makeschapterhead}[1]{%
  \vspace*{30pt}%
  {\parindent \z@ \centering
   \normalfont
   \usekomafont{chapter}#1\par\nobreak
   \vskip 40pt
  }}
\makeatother
